cmake_minimum_required(VERSION 3.19)
project(AetheriusDamageSystem LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(COMBAT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/server/server/cpp/server_guest_lib")
find_path(JSON_INCLUDE nlohmann/json.hpp HINTS "${CMAKE_CURRENT_SOURCE_DIR}/.deps/json/single_include")
if(NOT JSON_INCLUDE)
  message(FATAL_ERROR "nlohmann/json v3.11.3 required; set JSON_INCLUDE to its include directory")
endif()
add_library(aetherius_combat INTERFACE)
target_include_directories(aetherius_combat INTERFACE "${COMBAT_ROOT}" "${JSON_INCLUDE}")
add_executable(combat_tests tests/combat_tests.cpp)
target_link_libraries(combat_tests PRIVATE aetherius_combat)
target_compile_definitions(combat_tests PRIVATE COMBAT_CONFIG_DIR="${CMAKE_CURRENT_SOURCE_DIR}/server/server/config")
if(MSVC)
  target_compile_options(combat_tests PRIVATE /W4 /WX)
else()
  target_compile_options(combat_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
enable_testing()
add_test(NAME combat_core COMMAND combat_tests)
add_executable(golden_tests tests/golden/golden_tests.cpp)
target_link_libraries(golden_tests PRIVATE aetherius_combat)
target_compile_definitions(golden_tests PRIVATE
  COMBAT_CONFIG_DIR="${CMAKE_CURRENT_SOURCE_DIR}/server/server/config"
  GOLDEN_FIXTURE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/tests/golden/physical_damage_golden.json")
if(MSVC)
  target_compile_options(golden_tests PRIVATE /W4 /WX)
else()
  target_compile_options(golden_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME combat_golden COMMAND golden_tests)
find_path(PROMETHEUS_INCLUDE prometheus/core.h HINTS "${CMAKE_CURRENT_SOURCE_DIR}/.deps/prometheus/include")
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("#include <mutex>\nint main(){std::mutex m; m.lock(); m.unlock();}" COMBAT_HAS_STD_MUTEX)
if(PROMETHEUS_INCLUDE AND COMBAT_HAS_STD_MUTEX)
  add_executable(metrics_tests tests/metrics_tests.cpp)
  target_link_libraries(metrics_tests PRIVATE aetherius_combat)
  target_include_directories(metrics_tests SYSTEM PRIVATE "${PROMETHEUS_INCLUDE}")
  add_test(NAME combat_metrics COMMAND metrics_tests)
elseif(PROMETHEUS_INCLUDE)
  message(WARNING "Prometheus smoke test not built: this compiler lacks std::mutex. Native metrics require the Base's MSVC toolchain.")
endif()
